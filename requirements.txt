# requirements.txt — list of Python packages this project needs.
#
# Why this file exists:
# Your venv/ folder is intentionally excluded from GitHub (see .gitignore),
# because a virtual environment is tied to the exact machine it was built on.
# So when you pull this repo down on a different PC, that PC has an empty
# venv with nothing installed. This file is what lets you rebuild it with
# one command instead of remembering and reinstalling each package by hand.
#
# How to use it on a NEW machine (after cloning the repo and creating a venv):
#   1. source venv/bin/activate          (Mac/Linux)  or  venv\Scripts\activate  (Windows)
#   2. pip install -r requirements.txt
#
# These are the packages the application imports or runs directly. Pip will
# automatically pull in their smaller supporting packages (like jinja2 for
# Flask, or httpx for anthropic) on its own — you don't need to list those
# separately. The version numbers below are pinned to what's already
# installed and tested in your current venv, so installs stay predictable.

flask==3.1.3          # The web framework that runs the site and handles routes
itsdangerous==2.2.0   # Signs short-lived Interview AI follow-up context tokens
anthropic==0.112.0     # Claude API client, used by the /api/chat route
python-dotenv==1.2.2   # Loads your ANTHROPIC_API_KEY from the .env file
Flask-Limiter==4.1.1   # Adds basic abuse protection for the public chat endpoint
gunicorn==23.0.0       # The production web server Azure runs the app under.
                       # Azure's build system injected this invisibly before;
                       # pinning it here makes the dependency explicit and the
                       # app portable to any other host.
